# 2. استقبال اختيار الباقة وطلب معرف اللاعب (ID)
@bot.callback_query_handler(func=lambda call: call.data.startswith('buy_'))

    selected_pack = call.data.split('_')[1]
    
    user_orders[chat_id]['pack'] = selected_pack
    user_orders[chat_id]['price'] = PRICES[selected_pack]
    
    msg = bot.send_message(chat_id, "🎯 يرجى كتابة الـ ID الخاص بك في لعبة ببجي بدقة:")

# 5. معالجة ضغط أزرار القبول والرفض والربط بـ API نمر كارد
@bot.callback_query_handler(func=lambda call: call.data.startswith('adm_'))
def handle_admin_decision(call):
    data_parts = call.data.split('_')
    action = data_parts[1]
    client_chat_id = int(data_parts[2])
    
    if client_chat_id not in user_orders:
        bot.answer_callback_query(call.id, "⚠️ لم يتم العثور على بيانات هذا الطلب في الذاكرة.")
        return

    pack = user_orders[client_chat_id].get('pack')
    player_id = user_orders[client_chat_id].get('player_id')

    if action == "accept":
        # رابط إرسال الطلبات المباشر لمتجر نمر كارد 
        api_url = "https://nemer-card.com" 
        
        headers = {
            "Authorization": f"Bearer {NEMER_CARD_TOKEN}",
            "Content-Type": "application/json",
            "Accept": "application/json"
        }
        
        payload = {
            "product_id": PRODUCT_IDS.get(pack),
            "player_id": player_id,
            "quantity": 1
        }
        
        try:
            response = requests.post(api_url, json=payload, headers=headers, timeout=15)
            
            if response.status_code in:
                bot.edit_message_caption("✅ تم قبول الطلب وتمريره لمتجر نمر كارد للشحن التلقائي بنجاح.", call.message.chat.id, call.message.message_id)
                bot.send_message(client_chat_id, f"🎉 تم تأكيد الحوالة الخاصة بك! وجاري شحن {pack} شدة لحسابك (ID: {player_id}) تلقائياً. شكراً لثقتك بنا!")
            else:
                bot.send_message(ADMIN_CHAT_ID, f"⚠️ رد سيرفر نمر كارد بخطأ رقم {response.status_code}: {response.text}\nيرجى مراجعة رصيدك في الموقع كتاجر.")
                
        except Exception as e:
            bot.send_message(ADMIN_CHAT_ID, f"❌ فشل الاتصال التلقائي بمتجر نمر كارد بسبب: {e}")

    elif action == "reject":
        bot.edit_message_caption("❌ تم رفض هذا الطلب وإرسال التنبيه للمستخدم.", call.message.chat.id, call.message.message_id)
        bot.send_message(client_chat_id, "❌ نعتذر منك، لقد تم رفض طلب الشحن الخاص بك. يرجى التأكد من إرسال مبلغ الحوالة بشكل صحيح أو التواصل مع الدعم الفني.")

bot.infinity_polling()
